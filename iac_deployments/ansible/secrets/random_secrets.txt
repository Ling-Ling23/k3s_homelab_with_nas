


# View a specific vault file
ansible-vault view ansible/ansible_vault/homelab-ca-secret.yml

# View another vault file
ansible-vault view ansible_vault/k3s-secrets.yml

# Edit a vault file
ansible-vault edit ansible/ansible_vault/homelab-ca-secret.yml

# Decrypt to stdout (useful for piping)
ansible-vault decrypt --output=- ansible/ansible_vault/homelab-ca-secret.yml
